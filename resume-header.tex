% ============================================================
% Colors
% ============================================================

\definecolor{resumelink}{HTML}{28577A}
\definecolor{resumegray}{HTML}{666666}
\definecolor{sectionrule}{HTML}{B8B8B8}

\AtBeginDocument{%
  \hypersetup{
    colorlinks=true,
    urlcolor=resumelink,
    linkcolor=resumelink,
    citecolor=resumelink,
    pdfborder={0 0 0}
  }%

  % Force visible hyperlink text to use resumelink.
  \let\resumeoriginalhref\href
  \renewcommand{\href}[2]{%
    \resumeoriginalhref{#1}{\textcolor{resumelink}{#2}}%
  }%
}
